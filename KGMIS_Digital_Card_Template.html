/** * ============================================================ * KGMIS ADMINISTRATION MODULE * DIGITAL CARD ISSUANCE SERVICE * * Add this section at the bottom of: * 36.09_KGMIS_Module_Admin.gs * ============================================================ */ /** * Issues one Digital Membership Card selected through the * Administration Portal. * * The candidate is re-read from the database. No member details * supplied by the browser are trusted. * * @param {string} sessionToken * @param {string} candidateKey * @return {Object} */ function KGMIS_Admin_IssueDigitalCard( sessionToken, candidateKey ) { const user = KGMIS_OTP_RequireSessionAccess_( sessionToken, 'APPLICATION', 'ADMINISTER' ); const cleanCandidateKey = KGMIS_Admin_CleanText_(candidateKey); if (!cleanCandidateKey) { throw new Error( 'The selected member reference is missing.' ); } const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const candidate = KGMIS_Admin_FindCandidateByKey_( sessionToken, cleanCandidateKey ); KGMIS_Admin_ValidateCardIssueCandidate_( candidate ); const cardId = KGMIS_Admin_CreateCardIdForCandidate_( candidate ); const existingCard = KGMIS_ReadCardById_(cardId); if (existingCard) { throw new Error( 'Digital Card ' + cardId + ' has already been issued.' ); } const qrToken = KGMIS_Admin_CreateUniqueQrToken_(); const currentFinancialYear = KGMIS_GetCurrentFinancialYear(); if ( !currentFinancialYear || !currentFinancialYear.financialYear ) { throw new Error( 'The current Financial Year is not configured.' ); } const validUntil = KGMIS_Admin_GetCardValidityFromFinancialYear_( currentFinancialYear ); const now = new Date(); const issuedBy = KGMIS_Admin_CleanText_( user.email || user.userName || 'KGMIS_SYSTEM' ); const cardVersion = KGMIS_Admin_GetCurrentCardVersion_( currentFinancialYear ); const cardRecord = { CARD_ID: cardId, FAMILY_ID: KGMIS_Admin_NormaliseId_( candidate.familyId ), KEFG_ID: KGMIS_Admin_NormaliseId_( candidate.kefgId ), CARDHOLDER_TYPE: KGMIS_Admin_NormaliseCardholderType_( candidate.cardholderType ), RELATION_SEQUENCE: KGMIS_Admin_NormaliseSequence_( candidate.relationSequence ), CARDHOLDER_NAME: KGMIS_Admin_CleanText_( candidate.name ), MEMBERSHIP_TYPE: KGMIS_Admin_CleanText_( candidate.membershipType ).toUpperCase(), MEMBERSHIP_YEAR: KGMIS_Admin_CleanText_( candidate.membershipYear || currentFinancialYear.financialYear ), MEMBERSHIP_STATUS: KGMIS_Admin_NormaliseIssuedMembershipStatus_( candidate.membershipStatus ), ISSUE_DATE: now, VALID_UNTIL: validUntil, CARD_STATUS: 'ACTIVE', CARD_STATE: 'CURRENT', MEMBER_MOBILE: KGMIS_Admin_CleanText_( candidate.phone ), MEMBER_EMAIL: KGMIS_Admin_CleanText_( candidate.email ).toLowerCase(), PHOTO_FILE_ID: KGMIS_Admin_CleanText_( candidate.photoFileId ), PHOTO_URL: KGMIS_Admin_CleanText_( candidate.photoUrl ), QR_TOKEN: qrToken, CARD_PDF_FILE_ID: '', CARD_PDF_FILE_URL: '', CARD_IMAGE_FILE_ID: '', CARD_IMAGE_FILE_URL: '', CREATED_ON: now, CREATED_BY: issuedBy, UPDATED_ON: now, UPDATED_BY: issuedBy, CARD_VERSION: cardVersion, LAST_VERIFIED_ON: '', REMARKS: 'Issued through KGMIS Administration Portal.' }; const insertResult = KGMIS_InsertCard_( cardRecord ); return { success: true, message: 'Digital Membership Card issued successfully.', cardId: cardId, familyId: cardRecord.FAMILY_ID, kefgId: cardRecord.KEFG_ID, cardholderName: cardRecord.CARDHOLDER_NAME, cardholderType: cardRecord.CARDHOLDER_TYPE, membershipYear: cardRecord.MEMBERSHIP_YEAR, issueDate: KGMIS_Admin_FormatOutputDate_( cardRecord.ISSUE_DATE ), validUntil: KGMIS_Admin_FormatOutputDate_( cardRecord.VALID_UNTIL ), cardVersion: cardRecord.CARD_VERSION, qrToken: cardRecord.QR_TOKEN, databaseAction: insertResult && insertResult.action ? insertResult.action : 'INSERT' }; } finally { lock.releaseLock(); } } /** * Re-runs the existing Administration search and returns the * exact candidate matching candidateKey. */ function KGMIS_Admin_FindCandidateByKey_( sessionToken, candidateKey ) { const searchValue = KGMIS_Admin_GetCandidateSearchValue_( candidateKey ); const response = KGMIS_Admin_SearchDigitalCardCandidates( sessionToken, searchValue ); const results = response && Array.isArray(response.results) ? response.results : []; const normalizedCandidateKey = KGMIS_Admin_CleanText_( candidateKey ); const candidate = results.find(function (result) { return ( KGMIS_Admin_CleanText_( result.candidateKey ) === normalizedCandidateKey ); }); if (!candidate) { throw new Error( 'The selected member could not be found. ' + 'Please search again.' ); } return candidate; } /** * Derives a searchable database value from the candidate key. * * Supported examples: * KEFG:KEFG1001 * PERSON:DEP0001 * FAMILY_NAME:FAM00001:MEMBER NAME */ function KGMIS_Admin_GetCandidateSearchValue_( candidateKey ) { const cleanKey = KGMIS_Admin_CleanText_( candidateKey ); const separatorPosition = cleanKey.indexOf(':'); if (separatorPosition === -1) { return cleanKey; } const keyType = cleanKey .substring( 0, separatorPosition ) .toUpperCase(); const keyValue = cleanKey.substring( separatorPosition + 1 ); if ( keyType === 'KEFG' || keyType === 'PERSON' ) { return keyValue; } if (keyType === 'FAMILY_NAME') { const familySeparator = keyValue.indexOf(':'); return familySeparator === -1 ? keyValue : keyValue.substring( 0, familySeparator ); } return keyValue || cleanKey; } /** * Validates eligibility immediately before card insertion. */ function KGMIS_Admin_ValidateCardIssueCandidate_( candidate ) { if (!candidate) { throw new Error( 'A valid card candidate is required.' ); } if ( !KGMIS_Admin_NormaliseId_( candidate.familyId ) ) { throw new Error( 'The selected member has no FAMILY_ID.' ); } if ( !KGMIS_Admin_CleanText_( candidate.name ) ) { throw new Error( 'The selected member has no name.' ); } if ( !KGMIS_Admin_NormaliseSequence_( candidate.relationSequence ) ) { throw new Error( 'The selected member has no valid ' + 'RELATION_SEQUENCE.' ); } if (!candidate.cardEligible) { throw new Error( 'The selected person is not eligible for a Digital Card.' ); } if (candidate.cardIssued) { throw new Error( candidate.cardId ? 'Digital Card ' + candidate.cardId + ' has already been issued.' : 'A Digital Card has already been issued.' ); } if (!candidate.membershipYearRecordFound) { throw new Error( 'No ACTIVE membership-year record was found ' + 'for this family.' ); } const paymentStatus = KGMIS_Admin_NormaliseId_( candidate.paymentStatus ); if (paymentStatus !== 'PAID') { throw new Error( 'The family is not eligible because ' + 'PAYMENT_STATUS is not PAID.' ); } const membershipStatus = KGMIS_Admin_NormaliseText_( candidate.membershipStatus ); const eligibleStatuses = [ 'CURRENT', 'ACTIVE', 'EXTENDED', 'LIFE TIME', 'LIFETIME', 'LIFE-TIME' ]; if ( eligibleStatuses.indexOf( membershipStatus ) === -1 ) { throw new Error( 'The family has an ineligible ' + 'MEMBERSHIP_STATUS: ' + ( candidate.membershipStatus || 'BLANK' ) ); } } /** * Creates the deterministic Card Number: * * KEFG + five-digit family number + two-digit sequence */ function KGMIS_Admin_CreateCardIdForCandidate_( candidate ) { const familyId = KGMIS_Admin_NormaliseId_( candidate.familyId ); const match = familyId.match( /^FAM(\d{5})$/ ); if (!match) { throw new Error( 'Invalid FAMILY_ID format: ' + candidate.familyId + '. Expected FAM00001 format.' ); } const relationSequence = KGMIS_Admin_NormaliseSequence_( candidate.relationSequence ); if (!relationSequence) { throw new Error( 'Invalid RELATION_SEQUENCE.' ); } return ( 'KEFG' + match[1] + relationSequence ); } /** * Creates a unique QR token and verifies that it is not already * present in KEFG_MEMBER_CARDS. */ function KGMIS_Admin_CreateUniqueQrToken_() { const maximumAttempts = 5; for ( let attempt = 1; attempt <= maximumAttempts; attempt += 1 ) { const token = 'KGMIS-' + Utilities .getUuid() .replace(/-/g, '') .toUpperCase(); const existing = KGMIS_LookupCardByQrToken_( token ); if (!existing) { return token; } } throw new Error( 'A unique QR Token could not be generated.' ); } /** * Obtains the card validity date from the current Financial Year. */ function KGMIS_Admin_GetCardValidityFromFinancialYear_( currentFinancialYear ) { const value = currentFinancialYear.endDate || currentFinancialYear.END_DATE || currentFinancialYear.validUntil || ''; if ( Object.prototype.toString.call(value) !== '[object Date]' || isNaN(value.getTime()) ) { throw new Error( 'The current Financial Year has no valid END_DATE.' ); } return new Date( value.getTime() ); } /** * Obtains CARD_VERSION from the current financial-year record. */ function KGMIS_Admin_GetCurrentCardVersion_( currentFinancialYear ) { const version = KGMIS_Admin_CleanText_( currentFinancialYear.cardVersion || currentFinancialYear.CARD_VERSION ); if (!version) { throw new Error( 'CARD_VERSION is blank in the current ' + 'Financial Year record.' ); } return version; } /** * Returns the official cardholder type. */ function KGMIS_Admin_NormaliseCardholderType_( value ) { const type = KGMIS_Admin_NormaliseText_( value ); if ( [ 'PRIMARY MEMBER', 'MEMBER', 'DEPENDENT' ].indexOf(type) === -1 ) { throw new Error( 'Invalid CARDHOLDER_TYPE: ' + ( value || 'BLANK' ) ); } return type; } /** * The existing verification service expects CURRENT. * ACTIVE is therefore stored as CURRENT on a newly issued card. */ function KGMIS_Admin_NormaliseIssuedMembershipStatus_( value ) { const status = KGMIS_Admin_NormaliseText_( value ); if ( status === 'CURRENT' || status === 'ACTIVE' ) { return 'CURRENT'; } if (status === 'EXTENDED') { return 'EXTENDED'; } if ( status === 'LIFE TIME' || status === 'LIFETIME' || status === 'LIFE-TIME' ) { return 'LIFE TIME'; } throw new Error( 'Unsupported MEMBERSHIP_STATUS: ' + ( value || 'BLANK' ) ); } /** * Safe backend test. * * Do not run without replacing the two parameters with a valid * Administration session token and candidate key. */ function KGMIS_TestAdminIssueDigitalCard_( sessionToken, candidateKey ) { const result = KGMIS_Admin_IssueDigitalCard( sessionToken, candidateKey ); Logger.log( JSON.stringify( result, null, 2 ) ); return result; }
